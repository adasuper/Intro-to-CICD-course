// Alice's feature: true if player p holds a full row, column, or diagonal.
bool checkWin(const std::array<char, 9>& b, char p) {
    const int lines[8][3] = {
        {0,1,2}, {3,4,5}, {6,7,8},   // rows
        {0,3,6}, {1,4,7}, {2,5,8},   // columns
        {0,4,8}, {2,4,8}             // diagonals
    };
    for (auto& line : lines)
        if (b[line[0]] == p && b[line[1]] == p && b[line[2]] == p)
            return true;
    return false;
}
// Ben's feature: true once every cell holds a mark (no '1'..'9' left).
bool boardFull(const std::array<char, 9>& b) {
    for (char c : b)
        if (c != 'X' && c != 'O')
            return false;
    return true;
}

// tictactoe.cpp: end state (win + draw detection, bug fixed)
#include <iostream>
#include <array>

bool checkWin(const std::array<char, 9>& b, char p) {
    const int lines[8][3] = {
        {0,1,2}, {3,4,5}, {6,7,8},
        {0,3,6}, {1,4,7}, {2,5,8},
        {0,4,8}, {2,4,6}             // anti-diagonal fixed
    };
    for (auto& line : lines)
        if (b[line[0]] == p && b[line[1]] == p && b[line[2]] == p)
            return true;
    return false;
}

bool boardFull(const std::array<char, 9>& b) {
    for (char c : b)
        if (c != 'X' && c != 'O')
            return false;
    return true;
}

int main() {
    std::array<char, 9> board = {'1', '2', '3', '4', '5', '6', '7', '8', '9'};
    char player = 'X';

    auto draw = [&] {
        for (int r = 0; r < 3; ++r)
            std::cout << ' ' << board[r*3] << '|' << board[r*3+1] << '|' << board[r*3+2] << '\n';
    };

    while (true) {
        draw();
        std::cout << "Player " << player << ", cell 1-9: ";

        int cell;
        std::cin >> cell;
        if (cell < 1 || cell > 9 || board[cell-1] == 'X' || board[cell-1] == 'O') {
            std::cout << "invalid\n";
            continue;
        }

        board[cell-1] = player;
        if (checkWin(board, player)) { draw(); std::cout << "Player " << player << " wins!\n"; break; }
        if (boardFull(board))        { draw(); std::cout << "Draw\n"; break; }
        player = (player == 'X') ? 'O' : 'X';
    }
}
