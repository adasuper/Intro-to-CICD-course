The file you will track

When a step says to create tictactoe.py, paste the seed game below (a two-player hot-seat board; win detection arrives in Lab 2). For tracking purposes, write it in the table as tictactoe.py ("seed").

# tictactoe.py: seed (hot-seat, no win detection yet)

board = ['1', '2', '3', '4', '5', '6', '7', '8', '9']
player = 'X'
filled = 0


def draw():
    for r in range(3):
        print(f" {board[r*3]}|{board[r*3+1]}|{board[r*3+2]}")


while filled < 9:
    draw()
    cell = input(f"Player {player}, cell 1-9: ")

    if not cell.isdigit() or not (1 <= int(cell) <= 9) or board[int(cell) - 1] in ('X', 'O'):
        print("invalid")
        continue

    board[int(cell) - 1] = player
    filled += 1
    player = 'O' if player == 'X' else 'X'

draw()
print("Board full - draw (win detection in Lab 2)")
